#include <stdio.h>
#include <string.h>

struct date {
    int day, month, year;
};

struct address {
    char name[30];
    char city[20];
    int pincode;
};

struct parcel {
    int id;
    float weight;
    struct address sender;
    struct address receiver;
    struct date booked;
    int service;          /* 0 = normal, 1 = express */
    float charge;
};

void print_parcel(const struct parcel *p)
{
    printf("%d %.1f kg %s-%d %02d/%02d/%04d\n",
           p->id, p->weight, p->receiver.city,
           p->receiver.pincode, p->booked.day,
           p->booked.month, p->booked.year);
}

void sort_by_weight(struct parcel p[], int n)
{
    for (int i = 0; i < n - 1; i++) {
        int min = i;
        for (int j = i + 1; j < n; j++) {
            if (p[j].weight < p[min].weight)
                min = j;
        }
        if (min != i) {
            struct parcel temp = p[i];
            p[i] = p[min];
            p[min] = temp;
        }
    }
}

void list_by_city(const struct parcel p[], int n,
                  const char city[])
{
    for (int i = 0; i < n; i++) {
        if (strcmp(p[i].receiver.city, city) == 0)
            print_parcel(&p[i]);
    }
}

int main(void)
{
    struct parcel list[3] = {
        {501, 4.5f, {"Ravi", "Madurai", 625001},
         {"Meena", "Madurai", 625001}, {12, 10, 2026}, 0, 90.0f},
        {502, 1.2f, {"Kumar", "Chennai", 600077},
         {"Anu", "Chennai", 600077}, {12, 10, 2026}, 1, 120.0f},
        {503, 2.8f, {"Devi", "Coimbatore", 641001},
         {"Bala", "Coimbatore", 641001}, {13, 10, 2026}, 0, 75.0f}
    };

    sort_by_weight(list, 3);
    printf("Parcels sorted by weight:\n");
    for (int i = 0; i < 3; i++)
        print_parcel(&list[i]);

    printf("Parcels for receiver city Chennai:\n");
    list_by_city(list, 3, "Chennai");
    return 0;
}
