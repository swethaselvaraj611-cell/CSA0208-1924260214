#include <stdio.h>

struct account {
    int number;
    float balance;
};

/* Correct version: pointer refers to the original account. */
void addInterest(struct account *a, float rate)
{
    printf("Address inside function: %p\n", (void *)a);
    a->balance = a->balance + a->balance * rate / 100.0f;
}

int main(void)
{
    struct account list[3] = {
        {101, 5000.0f}, {102, 12000.0f}, {103, 800.0f}
    };

    struct account *p = list;

    for (int i = 0; i < 3; i++) {
        printf("Address in main:          %p\n",
               (void *)&list[i]);
        addInterest(&list[i], 4.0f);
    }

    printf("Account  Balance\n");
    for (int i = 0; i < 3; i++)
        printf("%d       %.2f\n",
               p[i].number, p[i].balance);

    printf("First account through pointer: %d %.2f\n",
           p->number, p->balance);
    return 0;
}
