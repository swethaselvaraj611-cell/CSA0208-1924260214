#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct { int id; char group[4]; } Donor;
typedef struct { Donor *data; size_t size, capacity; } DonorList;

int add_donor(DonorList *list, int id, const char *group)
{
    if (!list || !group || strlen(group) > 3) return 0;
    if (list->size == list->capacity) {
        size_t newCapacity = (list->capacity == 0) ? 2 : list->capacity * 2;
        Donor *temp = realloc(list->data, newCapacity * sizeof(*temp));
        if (temp == NULL) return 0; /* old allocation remains valid */
        list->data = temp;
        list->capacity = newCapacity;
    }
    list->data[list->size].id = id;
    strcpy(list->data[list->size].group, group);
    ++list->size;
    return 1;
}

void free_donor_list(DonorList *list)
{
    if (list) {
        free(list->data);
        list->data = NULL;
        list->size = 0;
        list->capacity = 0;
    }
}

int main(void)
{
    DonorList list = {NULL,0,0};
    const char *groups[] = {"O+","A+","B-","AB+","O-"};
    for (int i=0; i<5; ++i) {
        if (!add_donor(&list,7001+i,groups[i])) {
            fprintf(stderr,"Unable to allocate donor list\n");
            free_donor_list(&list);
            return 1;
        }
        printf("added %d (%s): size = %zu, capacity = %zu\n",
               list.data[i].id,list.data[i].group,list.size,list.capacity);
    }
    free_donor_list(&list);
    return 0;
}
