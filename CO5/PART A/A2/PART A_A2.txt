#include <stdio.h>

long long power_linear(long long base, int exp, long long *calls)
{
    ++(*calls);
    if (exp == 0) return 1;
    return base * power_linear(base, exp - 1, calls);
}

/* Computes base^exp using one recursive call for each halved exponent. */
long long power_fast(long long base, int exp, long long *calls)
{
    long long half;
    ++(*calls);
    if (exp == 0) return 1;
    half = power_fast(base, exp / 2, calls);
    if (exp % 2 == 0) return half * half;
    return base * half * half;
}

/* Repeated digit sum until one digit remains. */
int digital_root(int n)
{
    if (n < 10) return n;
    int sum = 0;
    while (n > 0) {
        sum += n % 10;
        n /= 10;
    }
    return digital_root(sum);
}

int main(void)
{
    long long c = 0;
    printf("digital root of 9875 = %d\n", digital_root(9875));
    printf("linear 2^10 = %lld, calls = ", power_linear(2,10,&c));
    printf("%lld\n", c);
    c = 0;
    printf("fast 2^10 = %lld, calls = ", power_fast(2,10,&c));
    printf("%lld\n", c);
    return 0;
}
