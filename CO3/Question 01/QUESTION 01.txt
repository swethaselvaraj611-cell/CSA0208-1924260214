PSEUDOCODE:

LINEAR_SEARCH(A, n, key)
    comparisons = 0
    FOR i = 0 TO n - 1
        comparisons = comparisons + 1
        IF A[i] = key
            RETURN i, comparisons
    RETURN -1, comparisons

BINARY_SEARCH(A, n, key)        // A must be sorted
    low = 0
    high = n - 1
    comparisons = 0
    WHILE low <= high
        mid = low + (high - low) / 2
        comparisons = comparisons + 1
        IF A[mid] = key
            RETURN mid, comparisons
        ELSE IF key < A[mid]
            high = mid - 1
        ELSE
            low = mid + 1
    RETURN -1, comparisons

PROGRAM:

#include <stdio.h>

int linear_steps, binary_steps;

int linear_search(const int a[], int n, int key)
{
    linear_steps = 0;
    for (int i = 0; i < n; i++) {
        linear_steps++;
        if (a[i] == key)
            return i;
    }
    return -1;
}

/* a[] must be sorted in ascending order. */
int binary_search(const int a[], int n, int key)
{
    int low = 0, high = n - 1;
    binary_steps = 0;

    while (low <= high) {
        int mid = low + (high - low) / 2;
        binary_steps++;

        if (a[mid] == key)
            return mid;
        else if (key < a[mid])
            high = mid - 1;
        else
            low = mid + 1;
    }
    return -1;
}

int main(void)
{
    int accession[] = {1021, 1048, 1103, 1190, 1222,
                       1305, 1377, 1402, 1460, 1533};
    int keys[] = {1021, 1305, 1533, 1400};
    int n = sizeof(accession) / sizeof(accession[0]);

    printf("LINEAR SEARCH (unsorted-array method)\n");
    for (int k = 0; k < 4; k++) {
        int pos = linear_search(accession, n, keys[k]);
        printf("Book %d -> index %d, comparisons = %d\n",
               keys[k], pos, linear_steps);
    }

    printf("\nBINARY SEARCH (sorted array)\n");
    for (int k = 0; k < 4; k++) {
        int pos = binary_search(accession, n, keys[k]);
        printf("Book %d -> index %d, comparisons = %d\n",
               keys[k], pos, binary_steps);
    }
    return 0;
}
