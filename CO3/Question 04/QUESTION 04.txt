#include <stdio.h>
#define DISTRICTS 38
#define MONTHS 12

int main(void)
{
    double rain[DISTRICTS][MONTHS];
    double annual[DISTRICTS] = {0};
    double monthlyTotal[MONTHS] = {0};
    double grandTotal = 0.0, maxRain = -1.0;
    int maxDistrict = 0, maxMonth = 0;

    printf("Enter rainfall (mm) for 38 districts, 12 months each:\n");

    for (int i = 0; i < DISTRICTS; i++) {
        printf("District %d: ", i + 1);
        for (int j = 0; j < MONTHS; j++) {
            if (scanf("%lf", &rain[i][j]) != 1 || rain[i][j] < 0) {
                printf("Invalid rainfall input.\n");
                return 1;
            }

            annual[i] += rain[i][j];
            monthlyTotal[j] += rain[i][j];
            grandTotal += rain[i][j];

            if (rain[i][j] > maxRain) {
                maxRain = rain[i][j];
                maxDistrict = i;
                maxMonth = j;
            }
        }
    }

    printf("\nANNUAL TOTAL BY DISTRICT\n");
    for (int i = 0; i < DISTRICTS; i++)
        printf("District %d: %.2f mm\n", i + 1, annual[i]);

    printf("\nSTATE MONTHLY AVERAGES\n");
    for (int j = 0; j < MONTHS; j++)
        printf("Month %d: %.2f mm\n", j + 1,
               monthlyTotal[j] / DISTRICTS);

    printf("\nWettest district-month: District %d, Month %d (%.2f mm)\n",
           maxDistrict + 1, maxMonth + 1, maxRain);

    double stateAnnualAverage = grandTotal / DISTRICTS;
    printf("State average annual rainfall: %.2f mm\n",
           stateAnnualAverage);
    printf("Districts below 60%% of state annual average:\n");

    for (int i = 0; i < DISTRICTS; i++) {
        if (annual[i] < 0.60 * stateAnnualAverage)
            printf("District %d: %.2f mm\n", i + 1, annual[i]);
    }

    return 0;
}
